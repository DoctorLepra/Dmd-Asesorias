-- 2. TABLA DE PERFILES (Profiles)
-- Se vincula automáticamente con auth.users
create table if not exists public.profiles (
  id uuid not null references auth.users on delete cascade primary key,
  full_name text,
  company text,
  contact_number text,
  role text default 'VISITOR' check (role in ('ADMINISTRATOR', 'EDITOR', 'VISITOR')),
  created_at timestamptz default now(),
  updated_at timestamptz default now()
);

-- Habilitar RLS
alter table public.profiles enable row level security;

-- Políticas de Perfiles
drop policy if exists "Perfiles son visibles por el dueño y administradores" on public.profiles;
create policy "Perfiles son visibles por el dueño y administradores"
  on public.profiles for select
  using ( auth.uid() = id or exists (select 1 from public.profiles where id = auth.uid() and role in ('ADMINISTRATOR', 'EDITOR')) );

drop policy if exists "Usuarios pueden actualizar su propio perfil" on public.profiles;
create policy "Usuarios pueden actualizar su propio perfil"
  on public.profiles for update
  using ( auth.uid() = id );

-- 3. TRIGGER PARA CREACIÓN AUTOMÁTICA DE PERFIL
-- Esta función se ejecuta cada vez que alguien se registra en Auth
create or replace function public.handle_new_user()
returns trigger as $$
begin
  insert into public.profiles (id, full_name, company, contact_number, role)
  values (
    new.id,
    new.raw_user_meta_data->>'full_name',
    new.raw_user_meta_data->>'company',
    new.raw_user_meta_data->>'contact_number',
    coalesce(new.raw_user_meta_data->>'role', 'VISITOR') -- Por defecto VISITOR
  );
  return new;
end;
$$ language plpgsql security definer;

-- Trigger
drop trigger if exists on_auth_user_created on auth.users;
create trigger on_auth_user_created
  after insert on auth.users
  for each row execute procedure public.handle_new_user();

-- 8. FUNCIONES RPC (Remote Procedure Calls) REQUERIDAS POR LA APP

-- Obtener todos los usuarios (UserManager.tsx)
-- Requiere security definer para acceder a auth.users
create or replace function public.get_all_users()
returns table (
  id uuid,
  email varchar,
  full_name text,
  company text,
  contact_number text,
  role text
)
language plpgsql security definer
as $$
begin
  if not exists (select 1 from public.profiles where id = auth.uid() and role in ('ADMINISTRATOR', 'EDITOR')) then
    raise exception 'Access denied';
  end if;

  return query
  select
    au.id,
    au.email::varchar,
    p.full_name,
    p.company,
    p.contact_number,
    p.role
  from auth.users au
  join public.profiles p on p.id = au.id;
end;
$$;

-- Obtener conversaciones calificadas (RatingsViewer.tsx)
create or replace function public.get_rated_conversations()
returns table (
  id uuid,
  created_at timestamptz,
  rating int,
  feedback text,
  full_name text,
  email varchar
)
language plpgsql security definer
as $$
begin
  if not exists (select 1 from public.profiles where id = auth.uid() and role in ('ADMINISTRATOR', 'EDITOR')) then
    raise exception 'Access denied';
  end if;

  return query
  select
    c.id,
    c.created_at,
    c.rating,
    c.feedback,
    p.full_name,
    au.email::varchar
  from public.ai_conversations c
  join public.profiles p on p.id = c.user_id
  join auth.users au on au.id = c.user_id
  where c.rating is not null
  order by c.created_at desc;
end;
$$;

-- Contar conversaciones calificadas (AIManager.tsx)
create or replace function public.count_rated_conversations()
returns int
language plpgsql security definer
as $$
begin
  if not exists (select 1 from public.profiles where id = auth.uid() and role in ('ADMINISTRATOR', 'EDITOR')) then
    raise exception 'Access denied';
  end if;
  return (select count(*) from public.ai_conversations where rating is not null);
end;
$$;

-- 9. STORAGE (Almacenamiento)
-- Nota: Los buckets deben crearse en la pestaña 'Storage' de Supabase,
-- pero las políticas se pueden definir aquí.

-- Asegurarse de que exista el bucket 'public_assets'
insert into storage.buckets (id, name, public) 
values ('public_assets', 'public_assets', true)
on conflict (id) do nothing;

-- Políticas Storage public_assets
drop policy if exists "Cualquiera puede ver public_assets" on storage.objects;
create policy "Cualquiera puede ver public_assets"
on storage.objects for select
using ( bucket_id = 'public_assets' );

drop policy if exists "Solo admin sube a public_assets" on storage.objects;
create policy "Solo admin sube a public_assets"
on storage.objects for insert
with check (
  bucket_id = 'public_assets' and
  exists (select 1 from public.profiles where id = auth.uid() and role in ('ADMINISTRATOR', 'EDITOR'))
);

drop policy if exists "Solo admin borra de public_assets" on storage.objects;
create policy "Solo admin borra de public_assets"
on storage.objects for delete
using (
  bucket_id = 'public_assets' and
  exists (select 1 from public.profiles where id = auth.uid() and role in ('ADMINISTRATOR', 'EDITOR'))
);

-- Asegurarse de que exista el bucket 'template_files'
